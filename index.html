កូដបញ្ជារថយន្តតាម Bluetooth & Ultrasonic 

#include <AFMotor.h>
#include <SoftwareSerial.h>

SoftwareSerial mySerial(9, 10); 
AF_DCMotor motor3(3); 
AF_DCMotor motor4(4); 

const int trigPin = A0; 
const int echoPin = A1; 

char bt = 'X'; 

void setup() { 
  Serial.begin(9600);   
  mySerial.begin(9600); 
  
  pinMode(trigPin, OUTPUT); 
  pinMode(echoPin, INPUT); 
  
  motor3.setSpeed(200); 
  motor4.setSpeed(200); 
}

void loop() {
  if (mySerial.available() > 0) {
    bt = mySerial.read(); 
  }

  if (Serial.available() > 0) {
    bt = Serial.read(); 
  }

  digitalWrite(trigPin, LOW); delayMicroseconds(2);
  digitalWrite(trigPin, HIGH); delayMicroseconds(10);
  digitalWrite(trigPin, LOW);
  
  long duration = pulseIn(echoPin, HIGH, 30000); 
  int distance = (duration == 0) ? 999 : (duration * 0.034 / 2);

  if (distance > 0 && distance < 10) {
    if (bt == 'F' || bt == 'f') {
      bt = 'X'; 
    }
  }

  switch (bt) {
    case 'F': case 'f': forward(); break;
    case 'B': case 'b': backward(); break;
    case 'L': case 'l': left(); break;
    case 'R': case 'r': right(); break;
    default: Stop(); break; 
  }
}

void forward() { motor3.run(FORWARD); motor4.run(FORWARD); }
void backward() { motor3.run(BACKWARD); motor4.run(BACKWARD); }
void left() { motor3.run(RELEASE); motor4.run(FORWARD); }
void right() { motor3.run(FORWARD); motor4.run(RELEASE); }
void Stop() { motor3.run(RELEASE); motor4.run(RELEASE); }
